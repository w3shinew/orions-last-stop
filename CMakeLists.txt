cmake_minimum_required(VERSION 3.20)
project(galactic2 VERSION 0.1.0 LANGUAGES CXX)
add_executable(galactic2 src/main.cpp)
target_include_directories(galactic2 PRIVATE include)
target_compile_features(galactic2 PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(galactic2 PRIVATE /W4 /permissive- /EHsc)
else()
  target_compile_options(galactic2 PRIVATE -Wall -Wextra -Wpedantic)
endif()
